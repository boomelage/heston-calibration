\documentclass[border=5pt]{standalone}
\usepackage{pgfplots}
\pgfplotsset{compat=1.18}

\begin{document}
\begin{tikzpicture}
    \begin{axis}[
        xlabel={Strike},
        ylabel={Price},
        colormap name=viridis,
        colorbar,
        point meta min=30,
        point meta max=720,
        colorbar style={ylabel={Days to maturity}},
    ]
        % All call maturities (30..720 days): strike (x) vs price (y),
        % each curve coloured by its days-to-maturity via the viridis map,
        % with a colorbar keyed by maturity.
        \foreach \t in {30,60,...,720} {
            \edef\plotcmd{%
                \noexpand\addplot[
                    mark=none,
                    color of colormap={(\t-30)/0.69},
                ] table[col sep=comma, x=strike, y=price] {%
                    ../data/surface_2020-3-16/plot_calls_\t-day_2020-3-16.csv};%
            }
            \plotcmd
        }
    \end{axis}
\end{tikzpicture}
\end{document}
