\documentclass[border=5pt]{standalone}
\usepackage{pgfplots}
\pgfplotsset{compat=1.18}
\usepgfplotslibrary{groupplots}
\usetikzlibrary{calc}

\begin{document}
% Data path differs by compile context: relative to this file when built
% standalone, relative to the repo root when \input into heston-calibration.tex.
\ifstandalone
    \def\smiledatadir{../data/surface_2020-3-16}
\else
    \def\smiledatadir{results/smiles/data/surface_2020-3-16}
\fi
\begin{tikzpicture}
    \begin{groupplot}[
        group style={
            group size=2 by 1,
            horizontal sep=8pt,
            ylabels at=edge left,
            yticklabels at=edge left,
        },
        width=7cm,
        height=7cm,
        colormap name=viridis,
        point meta min=30,
        point meta max=720,
        ymin=0, ymax=820,
    ]
        % Left: call maturities (30..720 days), strike (x) vs price (y),
        % each curve coloured by its days-to-maturity via the viridis map.
        \nextgroupplot[ylabel={Price}, title={Calls}]
        \foreach \t in {30,60,...,720} {
            \edef\plotcmd{%
                \noexpand\addplot[
                    mark=none,
                    color of colormap={(\t-30)/0.69},
                ] table[col sep=comma, x=strike, y=price] {%
                    \smiledatadir/plot_calls_\t-day_2020-3-16.csv};%
            }
            \plotcmd
        }
        % Right: put maturities, shared price axis, single common colorbar.
        \nextgroupplot[
            title={Puts},
            colorbar,
            colorbar style={ylabel={Days to maturity}},
        ]
        \foreach \t in {30,60,...,720} {
            \edef\plotcmd{%
                \noexpand\addplot[
                    mark=none,
                    color of colormap={(\t-30)/0.69},
                ] table[col sep=comma, x=strike, y=price] {%
                    \smiledatadir/plot_puts_\t-day_2020-3-16.csv};%
            }
            \plotcmd
        }
    \end{groupplot}
    % Single x-axis label centred between the two panels.
    \node[anchor=north, yshift=-1.1em]
        at ($(group c1r1.south west)!0.5!(group c2r1.south east)$)
        {Strike};
\end{tikzpicture}
\end{document}
