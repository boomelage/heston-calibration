		\begin{abstract}
			\noindent We specify a routine that calibrates the five canonical \citet{heston1993}
			parameters $\Phi=(\theta,\kappa,\eta,\rho,v_0)$ to a day of CBOE S\&P~500
			(SPX) intraday option trades. The routine performs \emph{one} calibration per
			trading day over a pooled, moneyness-normalised implied-volatility surface,
			and is framed throughout as a box-constrained nonlinear least-squares problem
			solved from multiple seeds. We make explicit every modelling choice: the
			volume-weighted reference spot $S_{\mathrm{ref}}$, the moneyness-normalised and
			grid-snapped strike $K^\star$, the surface-coverage requirements, the configurable
			in-engine objective (defaulting to implied-volatility ``IV-space'' residuals), the
			IV-space restart selection and acceptance gate, and the boundary-pegging rejection rule.
			The same routine optionally fits the \citet{bates1996} extension, which adds Merton lognormal
			jumps to the Heston dynamics, selected by a run-time flag and calibrated over the identical
			surface and gate. We keep Heston as the default throughout and present Bates as the
			eight-parameter generalisation it specialises from.
		\end{abstract}

	\section{The model}\label{sec:model}
		\subsection{Basis}
			Under the risk-neutral measure, with constant risk-free rate $r$ and continuous
			dividend yield $g$, the log-price $X_t=\ln S_t$ and instantaneous variance $v_t$ of the
			SPX follow the \citet{heston1993} dynamics
			\begin{align}
				dX_t &= \left( r - g - \tfrac{1}{2} v_t \right) dt
       			       	+ \sqrt{v_t}\,\bigl( \rho\, dW_t + \sqrt{1 - \rho^2}\, dB_t \bigr), \label{eq:level} \\
				dv_t &= \kappa(\theta - v_t)\, dt + \eta \sqrt{v_t}\, dW_t, \label{eq:variance}
			\end{align}
			where $W,B$ are independent Brownian motions, so $\mathrm{corr}(dX_t,dv_t)=\rho\,dt$.
			The parameter vector is
			\begin{equation}\label{eq:heston-params}
				\Phi = (\theta,\ \kappa,\ \eta,\ \rho,\ v_0) \in \Real^5,
			\end{equation}
			with $\theta>0$ the long-run variance, $\kappa>0$ the mean-reversion speed, $\eta>0$ the
			volatility of variance (QuantLib's \texttt{sigma}), $\rho\in[-1,1]$ the leverage
			correlation, and $v_0>0$ the initial variance. The Feller quantity \citep{feller1951}
			\begin{equation}\label{eq:feller}
				\mathcal{F}(\Phi) \;=\; 2\kappa\theta - \eta^2
			\end{equation}
			is reported per fit; $\mathcal{F}(\Phi)\ge 0$ guarantees $v_t>0$ almost surely. It is not
			imposed as a constraint (short-tenor SPX fits routinely violate it without being
			otherwise unreasonable; see \S\ref{sec:gate}).
	
		\subsection{Jumps}\label{sec:jumps}
			The \citet{bates1996} model extends Heston with \citet{merton1976} lognormal jumps in the
			log-price. The variance dynamics \eqref{eq:variance} are unchanged. The level equation
			\eqref{eq:level} gains a compound-Poisson jump term:
			\begin{equation}\label{eq:bates-level}
				dX_t = \left( r - g - \tfrac{1}{2} v_t - \lambda\,\xi \right) dt
				       + \sqrt{v_t}\,\bigl( \rho\, dW_t + \sqrt{1 - \rho^2}\, dB_t \bigr) + dJ_t .
			\end{equation}
			Here $J_t = \sum_{n=1}^{N_t} Y_n$ is a compound-Poisson process: $N_t$ is a Poisson process
			of intensity $\lambda\ge 0$ (expected jumps per year), independent of $W$ and $B$, and the
			log-jump sizes $Y_n \sim \mathcal{N}(\nu,\delta^2)$ are i.i.d. The compensator
			\begin{equation}\label{eq:bates-compensator}
				\xi = \mathbb{E}\bigl[ e^{Y} - 1 \bigr] = e^{\nu + \delta^2/2} - 1
			\end{equation}
			offsets the jump drift so the discounted price stays a martingale, leaving the level drift
			$r-g$ in expectation. The augmented parameter vector is
			\begin{equation}\label{eq:bates-params}
				\Theta = (\theta,\ \kappa,\ \eta,\ \rho,\ v_0,\ \lambda,\ \nu,\ \delta) \in \Real^8,
			\end{equation}
			the five Heston parameters of \eqref{eq:level}--\eqref{eq:variance} plus jump intensity
			$\lambda\ge 0$, mean log-jump $\nu$, and log-jump volatility $\delta>0$. At $\lambda=0$ the
			jump term vanishes and \eqref{eq:bates-level} collapses to the Heston level \eqref{eq:level},
			so Heston is the $\lambda=0$ face of the Bates parameter space. The model is selected at run
			time by a flag; the Heston fit is the default. The Feller quantity \eqref{eq:feller} is built
			from the diffusive parameters alone, so jumps do not enter it; it is reported under both models.

		\subsection{Pricing operators}\label{sec:pricing}
			Fix a valuation with maturity $\tau>0$ (in years), strike $K$, and type
			$w\in\{+1,-1\}$ ($+1$ call, $-1$ put). Write the forward and discount factor under flat
			curves as
			\begin{equation}\label{eq:forward}
				F(S,\tau) = S\, e^{(r-g)\tau}, \qquad D(\tau) = e^{-r\tau}.
			\end{equation}
		
			\textbf{Black price} \citep{black1976,blackscholes1973}. With total standard deviation $\Sigma=\sigma\sqrt{\tau}$,
			\begin{equation}\label{eq:black}
				\begin{gathered}
				C_{\mathrm{B}}(\sigma; S, K, \tau, w) = w\, D(\tau)\Bigl[ F\,\mathcal{N}(w d_+) - K\,\mathcal{N}(w d_-) \Bigr], \\
				d_\pm = \frac{\ln(F/K)}{\Sigma} \pm \tfrac{1}{2}\Sigma ,
				\end{gathered}
			\end{equation}
			where $\mathcal{N}$ is the standard normal CDF and $F=F(S,\tau)$.
		
			\textbf{Heston price.} The Heston model price $C_{\mathrm{H}}(\Phi; S,K,\tau,w)$ is
			the semi-analytic value
			\begin{equation}\label{eq:heston-price}
			\begin{gathered}
				C_{\mathrm{H}}(\Phi; S,K,\tau,+1) = S e^{-g\tau} P_1 - K e^{-r\tau} P_2, \\
				C_{\mathrm{H}}(\cdot,-1) = C_{\mathrm{H}}(\cdot,+1) - S e^{-g\tau} + K e^{-r\tau},
			\end{gathered}
			\end{equation}
			with the in-the-money probabilities, for $j\in\{1,2\}$,
			\begin{equation}\label{eq:heston-prob}
				P_j = \frac{1}{2} + \frac{1}{\pi}\int_0^\infty
				\Re\!\left[ \frac{e^{-i u \ln K}\, f_j(u; \Phi, \tau, \ln F)}{i u} \right] du ,
			\end{equation}
			and characteristic functions $f_j = \exp\bigl(C_j + D_j v_0 + i u \ln F\bigr)$ given by the standard coefficients
			\begin{equation}
				\begin{gathered}
					C_j = \frac{\kappa\theta}{\eta^2}\!\left[ (b_j - \rho\eta i u + d_j)\tau - 2\ln\!\frac{1 - \gamma_j e^{d_j\tau}}{1 - \gamma_j} \right],  \\
					D_j = \frac{b_j - \rho\eta i u + d_j}{\eta^2} \frac{1 - e^{d_j\tau}}{1 - \gamma_j e^{d_j\tau}},
				\end{gathered}
			\end{equation}
				\begin{equation}
				\begin{aligned}
					d_j &= \sqrt{(\rho\eta i u - b_j)^2 - \eta^2(2 \mu_j i u - u^2)},  \qquad \gamma_j &= \frac{b_j - \rho\eta i u + d_j}{b_j - \rho\eta i u - d_j},
				\end{aligned}
			\end{equation}
			where $\mu_1=\tfrac12,\ \mu_2=-\tfrac12,\ b_1=\kappa-\rho\eta,\ b_2=\kappa$. In code this
			integral is evaluated by the QuantLib \citep{quantlib} \texttt{AnalyticHestonEngine} (which uses a
			branch-stable formulation of \eqref{eq:heston-prob}); equations
			\eqref{eq:heston-price}--\eqref{eq:heston-prob} define the operator it realises.
		
			\textbf{Bates price.} The Bates price $C_{\mathrm{Bates}}(\Theta; S,K,\tau,w)$ has the same form
			as the Heston price \eqref{eq:heston-price}--\eqref{eq:heston-prob}, with each characteristic
			function $f_j$ multiplied by the Merton lognormal jump factor of \S\ref{sec:jumps}:
			\begin{equation}\label{eq:bates-cf}
				f_j^{\mathrm{B}}(u) = f_j(u)\,e^{\tau\,\psi_j(u)}, \qquad
				\psi_j(u) = \lambda\Bigl( e^{\,i u_j \nu - \frac12 \delta^2 u_j^2} - 1 \Bigr) - i u_j\,\lambda\xi,
			\end{equation}
			with $u_2 = u$, $u_1 = u - i$ (the same measure shift that distinguishes $f_1$ from $f_2$), and
			$\xi = e^{\nu + \delta^2/2} - 1$ the compensator \eqref{eq:bates-compensator}. The exponent $\psi_j$
			is the Merton log-jump cumulant, compensated by $-i u_j \lambda\xi$ so the drift correction of
			\eqref{eq:bates-level} carries through. At $\lambda=0$ the factor is $1$ and $C_{\mathrm{Bates}}$
			reduces to $C_{\mathrm{H}}$. In code the integral is evaluated by the QuantLib \texttt{BatesEngine},
			the jump-augmented counterpart of the \texttt{AnalyticHestonEngine}.
		
			\textbf{Implied volatility.} The Black-implied volatility of a price $C$ is the inverse
			of \eqref{eq:black} in its first argument,
			\begin{equation}\label{eq:ivinv}
				\mathcal{I}(C; S,K,\tau,w)
				= \bigl\{\, \sigma>0 : C_{\mathrm{B}}(\sigma; S,K,\tau,w) = C \,\bigr\},
			\end{equation}
			computed numerically (QuantLib \texttt{blackFormulaImpliedStdDev} on the forward
			\eqref{eq:forward}, returning $\Sigma$, then $\sigma=\Sigma/\sqrt{\tau}$). This operator
			defines the IV-space residuals of \S\ref{sec:objective}.
	
	\section{Data and per-day notation}\label{sec:data}
		The pipeline ingests CBOE trade-level files, one per trading day $d$, restricted upstream
		to out-of-the-money (OTM) quotes. A day is a set of trade records indexed by $i$, each
		carrying
		\begin{equation}
			\bigl(t_i,\ K_i,\ w_i,\ q_i,\ P_i,\ \sigma_i,\ S_i,\ \tau_i\bigr),
		\end{equation}
		being respectively the quote timestamp, strike, type $w_i\in\{+1,-1\}$, traded size, trade
		price, trade-implied volatility, the underlying level \emph{at trade time} $t_i$, and the
		days to maturity. Maturities enter the engine as $\tau_i$ in years under
		$\mathrm{Actual}/365$, i.e.\ $T_i = \tau_i/365$.
	
		The day is rate-stamped with $r=r(d)$ and $g=g(d)$, the most recent risk-free and
		dividend rates on or before $d$ (a flat-forward curve for each; under flat curves the
		evaluation date is immaterial to $T_i$, so it is set to the QuantLib ``today'').
	
		\textbf{Pre-filter.} We keep only economically usable, Heston-appropriate quotes. Write the
		ratio moneyness $\varrho_i = e^{-\lvert \ln(K_i/S_i)\rvert}\in(0,1]$ (it equals $1$ at the money
		and falls as the strike moves out). The kept set is
		\begin{equation}\label{eq:prefilter}
			\mathcal{D}
			= \bigl\{\, i \;:\; \sigma_i > 0,\ \ \tau_{\min} \le \tau_i \le \tau_{\max},\ \
			  \underline{m} < \varrho_i < \bar m \,\bigr\}.
		\end{equation}
		The maturity floor $\tau_{\min}$ (Table~\ref{tab:const}) drops ultra-short maturities, which the
		diffusive Heston smile fits poorly and which otherwise drive $\eta,\kappa$ to Feller-violating
		extremes that pollute the pooled fit; the ceiling $\tau_{\max}$ drops thin, stale long tenors.
		The moneyness band keeps only out-of-the-money quotes: the cutoff $\bar m$ removes the near-ATM
		rows, and the floor $\underline{m}$ removes the deep-OTM lottery-ticket tail
		($\varrho_i<\underline{m}$, roughly $\lvert\ln(K_i/S_i)\rvert>0.51$), whose extreme prices peg the
		fit to its bounds. Flooring that tail recovers acceptance and tightens the fit while keeping the
		full tradeable wing.
	
	\section{Handling intraday spot movement}\label{sec:spot}
		A Heston calibration is anchored to a \emph{single} spot $S$, yet the trades in
		$\mathcal{D}$ occur across an intraday range of underlying levels $\{S_i\}$. We reconcile
		the two with one reference spot and a moneyness re-striking.
	
		\textbf{Reference spot.} Define the volume-weighted reference spot
		\begin{equation}\label{eq:sref}
			S_{\mathrm{ref}} \;=\; \frac{\sum_{i\in\mathcal{D}} q_i\, S_i}{\sum_{i\in\mathcal{D}} q_i}.
		\end{equation}
		The intraday range and its flag are
		\begin{equation}\label{eq:highmove}
			R \;=\; \frac{\max_{i} S_i}{\min_{i} S_i} - 1,
			\qquad
			\textsc{high\_move} \;=\; \mathbf{1}\{ R > R_{\max} \}.
		\end{equation}
		A flagged day ($R>R_{\max}$) is still calibrated and written, but the normalisation below
		is recorded as strained (its $S_{\mathrm{ref}}$ should be treated with suspicion).
	
		\textbf{Moneyness normalisation and grid snap.} Each trade retains its moneyness at
		trade time and is re-struck to the reference spot, then snapped to the SPX strike grid of
		width $\Delta$:
		\begin{equation}\label{eq:kstar}
			m_i = \frac{K_i}{S_i},
			\qquad
			K^\star_i \;=\; \Delta \cdot \round\!\left( \frac{m_i\, S_{\mathrm{ref}}}{\Delta} \right).
		\end{equation}
		The rationale is sticky-moneyness: over a session the implied-volatility smile is
		approximately stationary in $K/S$ rather than in the absolute strike. Re-striking by
		\eqref{eq:kstar} therefore re-centres the whole day onto $S_{\mathrm{ref}}$ so that trades
		executed at different $S_i$ but the \emph{same} moneyness share a common surface column.
		The grid snap collapses the continuum of $K^\star_i$ values (one per trade) back onto a
		clean strike lattice. Pricing diagnostics (\S\ref{sec:output}) nonetheless use each
		contract's \emph{original} $(S_i,K_i)$, since the calibrated parameters are spot-independent.

	\section{Surface construction}\label{sec:surface}
		From the normalised trades we build a partial $K^\star \times \tau$ implied-volatility
		surface, selecting the most liquid, nearest-the-money quotes.
	
		\textbf{Maturity selection.} For each distinct maturity $\tau$ present in $\mathcal{D}$,
		let $V(\tau)=\sum_{i\in\mathcal{D}:\,\tau_i=\tau} q_i$ be its traded volume. Keep the
		$N_t$ highest-volume maturities,
		\begin{equation}
			\mathcal{T} = \text{top-}N_t \text{ maturities by } V(\tau).
		\end{equation}
	
		\textbf{Strike selection (per maturity, per wing).} Within each kept maturity
		$\tau\in\mathcal{T}$, separate calls ($w_i=+1$, OTM with $K^\star_i \gtrsim S_{\mathrm{ref}}$)
		from puts ($w_i=-1$, OTM with $K^\star_i \lesssim S_{\mathrm{ref}}$) and keep the $n_k$
		nearest-the-money distinct strikes on each wing:
		\begin{equation}
			\mathcal{K}_\tau \;=\;
			\underbrace{\bigl\{\text{$n_k$ largest } K^\star \text{ among puts at } \tau\bigr\}}_{\text{put wing}}
			\ \cup\
			\underbrace{\bigl\{\text{$n_k$ smallest } K^\star \text{ among calls at } \tau\bigr\}}_{\text{call wing}} .
		\end{equation}
	
		\textbf{Pivot to a surface.} For each retained cell $(k,\tau)$ with $k\in\mathcal{K}_\tau$,
		set the surface entry to the implied volatility of the \emph{highest-volume} trade in that
		cell, the one with the largest traded size,
		\begin{equation}\label{eq:pivot}
			\hat\sigma_{k,\tau} \;=\; \sigma_{i^\star}, \qquad
			i^\star = \argmax_{i\,:\,K^\star_i=k,\ \tau_i=\tau}\ q_i ,
		\end{equation}
		and leave it missing otherwise. When several trades tie at the maximum size the pick is
		arbitrary. Let
		\begin{equation}
			\mathcal{N} = \bigl\{(k,\tau) : \hat\sigma_{k,\tau}\ \text{defined}\bigr\}
		\end{equation}
		be the set of populated cells; these become the calibration targets (helpers).
	
		\textbf{Coverage requirement.} To keep the five-parameter fit identified we require a
		genuinely two-dimensional surface:
		\begin{equation}\label{eq:coverage}
			\bigl|\{\tau : (\cdot,\tau)\in\mathcal{N}\}\bigr| \ge M_t,
			\quad
			\bigl|\{k : (k,\cdot)\in\mathcal{N}\}\bigr| \ge M_k,
			\quad
			|\mathcal{N}| \ge M_{\mathrm{cell}} .
		\end{equation}
		A day failing \eqref{eq:coverage} (or with no usable maturities) is dropped: it
		contributes no calibration and clears its stale diagnostics.

	\section{The optimization problem}\label{sec:objective}
		The routine calibrates one parameter vector per day. Under the default Heston model this is $\Phi$
		of \eqref{eq:heston-params} with $\dim\Phi=5$; under Bates it is $\Theta$ of \eqref{eq:bates-params} with
		$\dim\Theta=8$. The two are nested. They share their first $\dim\Phi$ coordinates, the diffusive
		parameters $(\theta,\kappa,\eta,\rho,v_0)$, and $\Theta$ appends the jump triple
		$(\lambda,\nu,\delta)$. Setting $\lambda=0$ collapses the jump term \eqref{eq:bates-level} and
		recovers $\Phi$, so Heston is the $\lambda=0$ face of Bates. We write $\Psi$ for the active vector,
		either $\Phi$ or $\Theta$, and $C^{\mathrm{mod}}(\Psi;\cdot)$ for the active model price: the Heston
		price $C_{\mathrm{H}}$ \eqref{eq:heston-price} when $\Psi=\Phi$, and the Bates price
		$C_{\mathrm{Bates}}$ \eqref{eq:bates-cf} when $\Psi=\Theta$. Everything below is written in $\Psi$
		and holds for either model. The only place the model enters is the box dimension $\dim\Psi$
		\eqref{eq:box} and, through it, the bounds tables.
	
		Let $\mathcal{N}$ index the targets with market data $\{(\hat\sigma_n, K^\star_n, T_n, w_n)\}_{n\in\mathcal{N}}$,
		all priced at the single spot $S_{\mathrm{ref}}$. Define the feasible box
		\begin{equation}\label{eq:box}
			\mathcal{B} = \prod_{j=1}^{\dim\Psi} [\ell_j, h_j] \subset \Real^{\dim\Psi},
		\end{equation}
		with $(\ell,h)$ in the \texttt{model.params()} order of Table~\ref{tab:bounds}: the five Heston
		coordinates, extended by the three jump bounds when $\Psi=\Theta$. The $\lambda$ floor is exactly
		$0$, so the Bates box contains the Heston model on its $\lambda=0$ face.

		\subsection{Residuals}
			Two residual systems are defined: an IV-space system and a price-space system. The in-engine
			objective is configurable. It defaults to the IV-space system \eqref{eq:obj-iv} below, and can
			instead be set to the price-space system \eqref{eq:resid-price}. The choice changes only what
			each restart minimises. Selection \eqref{eq:select} and the acceptance gate \eqref{eq:accept}
			always run on the IV-space error $J_{\mathrm{iv}}$, so under the default objective the optimizer
			minimises the very error that then selects and gates.
	
			\textbf{Price-space (alternative in-engine objective).} For each target the market value is the
			Black price at the quoted vol, $C^{\mathrm{mkt}}_n = C_{\mathrm{B}}(\hat\sigma_n; S_{\mathrm{ref}}, K^\star_n, T_n, w_n)$,
			and the relative-price residual is
			\begin{equation}\label{eq:resid-price}
				\varepsilon^{\mathrm{px}}_n(\Psi)
				= \frac{C^{\mathrm{mod}}(\Psi; S_{\mathrm{ref}}, K^\star_n, T_n, w_n) - C^{\mathrm{mkt}}_n}{C^{\mathrm{mkt}}_n}.
			\end{equation}
			The associated objective is the root-mean-square
			\begin{equation}\label{eq:obj-price}
				J_{\mathrm{px}}(\Psi)
				= \left( \frac{1}{|\mathcal{N}|} \sum_{n\in\mathcal{N}} \varepsilon^{\mathrm{px}}_n(\Psi)^2 \right)^{1/2}.
			\end{equation}
			The per-target residual $\varepsilon^{\mathrm{px}}_n$ is the (relative-price) error each
			QuantLib helper exposes. Levenberg--Marquardt \citep{marquardt1963} is a sum-of-squares
			method, so it drives the squared residual norm
			$\sum_{n}(\varepsilon^{\mathrm{px}}_n)^2 = |\mathcal{N}|\,J_{\mathrm{px}}(\Psi)^2$ to a local
			minimum (equivalently $J_{\mathrm{px}}$, with which it shares a minimiser).
		
			\textbf{IV-space (default in-engine objective; selection and gate).} Invert each model price back to a Black vol via
			\eqref{eq:ivinv} and compare to the market vol:
			\begin{equation}\label{eq:resid-iv}
			\begin{gathered}
				\sigma^{\mathrm{mod}}_n(\Psi) = \mathcal{I}\bigl( C^{\mathrm{mod}}(\Psi; S_{\mathrm{ref}}, K^\star_n, T_n, w_n);
				    S_{\mathrm{ref}}, K^\star_n, T_n, w_n \bigr), \\
				\varepsilon^{\mathrm{iv}}_n(\Psi) = \sigma^{\mathrm{mod}}_n(\Psi) - \hat\sigma_n,
			\end{gathered}
			\end{equation}
			with objective
			\begin{equation}\label{eq:obj-iv}
				J_{\mathrm{iv}}(\Psi)
				= \left( \frac{1}{|\mathcal{N}'|} \sum_{n\in\mathcal{N}'} \varepsilon^{\mathrm{iv}}_n(\Psi)^2 \right)^{1/2},
				\qquad \mathcal{N}' = \{ n : \sigma^{\mathrm{mod}}_n(\Psi)\ \text{inverts finitely} \}.
			\end{equation}
			$J_{\mathrm{iv}}$ is measured in volatility points; it is the natural surface-fit metric and,
			unlike $J_{\mathrm{px}}$, is not inflated by cheap deep-OTM wings (a unit price error on a
			near-worthless contract is a large \emph{relative} price error but a small vol error).
	
		\subsection{Multi-start local optimization}\label{sec:multistart}
			A single start into a non-convex problem is fragile \citep{mikhailov2003}, so we restart from a small,
			data-seeded set. Let the surface variance level be
			\begin{equation}\label{eq:varproxy}
				\hat v = \clip\Bigl( \!\! \bigl(\med_{(k,\tau) \in \, \mathcal{N}} \hat\sigma_{k,\tau}\bigr)^2,\ [10^{-3},\ 0.25] \Bigr),
			\end{equation}
			and define the seed set $\mathcal{S}=\{s_1,\dots,s_6\}\subset\mathcal{B}$ of
			$(v_0,\kappa,\theta,\eta,\rho)$ tuples
			\begin{equation}\label{eq:seeds}
				\mathcal{S} =
				\begin{cases}
					(\hat v,\ 1.0,\ \hat v,\ 0.50,\ -0.70), &
					(\hat v,\ 3.0,\ \hat v,\ 1.00,\ -0.50), \\
					(\hat v,\ 0.5,\ \hat v,\ 0.30,\ -0.90), &
					(\hat v,\ 5.0,\ \hat v,\ 0.80,\ -0.60), \\
					(0.8\hat v,\ 2.0,\ 1.2\hat v,\ 0.60,\ -0.75), &
					(\hat v,\ 8.0,\ \hat v,\ 1.20,\ -0.40).
				\end{cases}
			\end{equation}
			Under Bates each seed is extended by one fixed jump triple
			$(\lambda,\nu,\delta)=(0.1,-0.1,0.1)$, so every restart starts from
			almost no jumps and grows them only if they tighten the fit; the restart count is unchanged.
			For each seed $s\in\mathcal{S}$ we solve the box-constrained nonlinear least-squares problem
			\begin{equation}\label{eq:localproblem}
				\widehat\Psi_s \;=\; \argmin_{\Psi \in \mathcal{B}}\ \sum_{n\in\mathcal{N}} \varepsilon^{\mathrm{obj}}_n(\Psi)^2
				\qquad \text{(LM, started at } \Psi^{(0)}=s\text{)},
			\end{equation}
			using Levenberg--Marquardt under the box constraint $\mathcal{B}$ (QuantLib \texttt{\seqsplit{NonhomogeneousBoundaryConstraint}}). 
			The per-target residual $\varepsilon^{\mathrm{obj}}_n$ is the configured in-engine objective: 
			by default the IV-space residual $\varepsilon^{\mathrm{iv}}_n$ \eqref{eq:resid-iv}, optionally 
			the price-space residual $\varepsilon^{\mathrm{px}}_n$ \eqref{eq:resid-price}. The fit retained 
			for the day is the restart with the smallest \emph{IV-space} error,
			\begin{equation}\label{eq:select}
				\Psi^\star \;=\; \argmin_{s\in\mathcal{S}}\ J_{\mathrm{iv}}\!\bigl(\widehat\Psi_s\bigr).
			\end{equation}
			Ranking by $J_{\mathrm{iv}}$ (not $J_{\mathrm{px}}$) removes the dependence on any single
			arbitrary guess and selects the economically best surface fit.

	\section{Acceptance gate}\label{sec:gate}
		$\Psi^\star$ is accepted only if it fits the surface tightly \emph{and} lies in the interior
		of the box. Let the boundary-pegging predicate, with tolerance fraction $\delta$, be
		\begin{equation}\label{eq:pegged}
			\textsc{pegged}(\Psi)
			= \bigvee_{j=1}^{\dim\Phi}
			\Bigl[\ (\Psi_j - \ell_j) < \delta\,(h_j-\ell_j)
			\ \ \vee\ \
			(h_j - \Psi_j) < \delta\,(h_j-\ell_j) \ \Bigr].
		\end{equation}
		The disjunction runs over the $\dim\Phi$ diffusive coordinates for both models. Under Bates the three jump
		coordinates $(\nu,\delta,\lambda)$ are exempt from the predicate: $\lambda\to0$ is a legitimate
		collapse to Heston rather than the optimizer hitting a wall, and when $\lambda\approx0$ the jump
		shape $(\nu,\delta)$ is unidentified and may rest on a bound without meaning. The jump triple is
		still reported. The Feller quantity \eqref{eq:feller} likewise stays the diffusive quantity under
		Bates and never gates.
		The day is accepted iff
		\begin{equation}\label{eq:accept}
			\boxed{\;
			J_{\mathrm{iv}}(\Psi^\star) \le \varepsilon_{\mathrm{iv}}
			\quad\text{and}\quad
			\lnot\,\textsc{pegged}(\Psi^\star).
			\;}
		\end{equation}
		The first clause is the IV-space gate ($\varepsilon_{\mathrm{iv}}\approx 2$ vol points). The
		second rejects boundary fits: a parameter resting on a bound is the optimizer hitting a wall,
		not an identified value, so it is treated as a non-fit regardless of $J_{\mathrm{iv}}$. The
		Feller quantity \eqref{eq:feller} is reported but never gates, since short-tenor SPX fits can
		violate it while remaining otherwise sound. A rejected or too-thin day yields no parameter
		row.
	
		\textbf{Why an IV-space gate.} An earlier version gated on $J_{\mathrm{px}}$ directly. On a
		pooled day rich in deep-OTM wings, $J_{\mathrm{px}}$ is dominated by those cheap contracts and
		rejects fits that are excellent in vol points. Replacing the gate metric with
		$J_{\mathrm{iv}}$ aligns the acceptance criterion with the units the surface is quoted in;
		under it the genuine pooled-surface fit is $\approx 0.8$--$1.4$ vol points, and the residual
		rejections are boundary-pegged $\kappa$ or $\rho$ (a separate identification lever), not
		fit-quality failures.

	\section{Output and diagnostics}\label{sec:output}
		An accepted day yields one parameter row keyed by date,
		\begin{equation}
			\bigl(d,\ S_{\mathrm{ref}},\ r,\ g,\ \theta^\star,\kappa^\star,\rho^\star,\eta^\star,v_0^\star,\
			\mathcal{F}(\Psi^\star),\ J_{\mathrm{iv}},\ J_{\mathrm{px}},\ \dots,\ R,\ \textsc{high\_move}\bigr),
		\end{equation}
		together with coverage counts $(|\mathcal{T}|, \#\text{strikes}, |\mathcal{N}|)$ and total
		volume. Under Bates the row appends the three jump parameters $(\lambda^\star,\nu^\star,\delta^\star)$
		after $v_0^\star$, and the repriced diagnostics carry a Bates price column in place of the Heston
		one; everything else is identical. Outputs are routed per model and objective, so the Heston and
		Bates rows never share a file. Because the calibration is once-daily, these rows accumulate into a
		single file, regenerated each run from the accepted days. For repricing diagnostics each retained contract
		is valued under $\Psi^\star$ at its \emph{original} $(S_i,K_i)$, not $(S_{\mathrm{ref}},K^\star)$,
		since the structural parameters are spot-independent; this gives an honest comparison against
		the actual trade price $P_i$.

	\section{Empirical findings}\label{sec:empirical}
		All findings in this section are for the Heston model; a full-sample Bates run is left to later work
		(\S\ref{sec:improvements}). We ran the routine over the full CBOE SPX trade history available to us,
		$3{,}217$ trading days from 3~January~2012 to 15~October~2024. The figures below are for
		the default IV-space objective \eqref{eq:obj-iv}. The price-space objective
		\eqref{eq:obj-price} is the configurable alternative, discussed at the end of the section.
		Of these days, $1{,}631$ ($50.7\%$) passed the acceptance gate \eqref{eq:accept} and
		contribute a parameter row; the $1{,}586$ rejected days are logged with their cause.
		The rejection split is dominated by boundary pegging: $1{,}467$ days
		($92.5\%$ of rejections) are rejected for a pegged parameter, $116$ for an IV-space miss
		($J_{\mathrm{iv}}>\varepsilon_{\mathrm{iv}}$), and $3$ for having no usable trades.
		Pegging (almost always $\kappa\!\to\!20$ or $\rho\!\to\!-0.999$ hitting the box wall)
		is therefore the binding constraint on the accept rate, not surface quality.
		\\
		\textbf{Fit quality.} On the accepted set the surface fit is excellent. The IV-space
		error \eqref{eq:obj-iv} has a median of $0.80$ vol points (mean $0.84$, $90$th percentile
		$1.15$, maximum $1.99$), so $J_{\mathrm{iv}}$ sits inside the $2$-vol-point gate on every
		accepted day and the gate never binds in practice. Acceptance is governed by the no-pegging
		clause alone. An independent repricing check (each contract valued under $\Phi^\star$ at its
		\emph{original} $(S_i,K_i)$ and compared to the trade price) gives a per-day median relative
		price error of $11.4\%$ with a $90$th percentile of $17.4\%$. Those figures are inflated by the
		deep-OTM wings, whose near-zero prices make any absolute miss a large relative one. This is
		exactly the effect that motivated gating in IV space rather than price space
		(\S\ref{sec:gate}). A separate validator clears $1{,}623$ of $1{,}631$ accepted days
		($99.5\%$); the $8$ hard failures are repricing-fit outliers, not parameter pathologies.
		\\ 
		\textbf{Parameter levels.} The accepted parameters are economically sensible and stable
		across the 2012--2024 sample. The leverage correlation is consistently and strongly
		negative, $\med\rho=-0.74$ (10th--90th percentile $[-0.81,-0.61]$), as equity-index
		smiles require. Long-run variance $\med\theta=0.067$ ($\approx 25.8\%$ annualised vol) and
		initial variance $\med v_0=0.026$ ($\approx 16.2\%$) bracket realistic SPX levels, with $v_0$
		the lower of the two on the typical day. Mean reversion $\med\kappa=2.37$ sits in the usual
		SPX range. The vol-of-vol is high, $\med\eta=1.34$ (10th--90th percentile $[0.79,1.84]$).
		Because the gate rejects any boundary-pegged fit, the accepted set is pegging-free by
		construction: no accepted day carries $\rho$ below $-0.99$ or reaches the $\kappa$ ceiling.
		A clean interior set is what survives the gate, not evidence that pegging is solved on the
		underlying population.
		\\ 
		\textbf{Feller condition is the standout caveat.} The gate does not test Feller, and these
		fits violate it without exception: $\mathcal{F}(\Phi^\star)=2\kappa\theta-\eta^2<0$ on all
		$1{,}631$ accepted days, with median $\mathcal{F}=-1.44$. The driver of the violation is the
		large vol-of-vol relative to $\kappa\theta$. $\eta$ exceeds $1.5$ on $596$ days ($36.5\%$) and
		approaches its $2.0$ cap (max $1.997$). A pinned $v_0$ or a degenerate $\kappa$ are rare, so
		among the validator's soft flags Feller violation and an elevated $\eta$ are by far the most
		common. Together they mark these fits as ones whose variance path can reach zero, even where
		the surface fit itself is tight. The wide surface coverage is what lifts $\eta$ here: putting
		the full OTM wings into the fit demands more vol-of-vol than a near-money fit, so both $\eta$
		and the Feller violation run higher than under the narrower near-money calibration of earlier
		drafts.
		\\ 
		\textbf{Comparison with the earlier price-space baseline.} An earlier draft reported the
		price-space objective on a narrower surface (near-money strikes only, a higher maturity floor)
		as the default. That run accepted $1{,}713$ of $3{,}215$ days ($53.3\%$) with a lower vol-of-vol
		($\med\eta\approx0.89$, $\eta>1.5$ on $8.5\%$ of days) and Feller violated on $99.3\%$. Two
		things changed the picture here. The default objective is now the IV-space residual, and the
		surface coverage was widened to carry the full tradeable wing and longer maturities, with a
		deep-OTM floor that drops the lottery-ticket tail. The widening is the larger effect. It removes
		what looked like a systematic wing underfit: with the wings in the fit, the per-$\lvert\ln(K/S)\rvert$
		residual is small and mixed in sign (overall RMSE $1.1$ vol points), so the near-money-only
		calibration was extrapolating, not capturing an in-sample bias. The same widening raises $\eta$
		and pushes Feller violation to every accepted day. The accept rate is little changed
		($50.7\%$ against $53.3\%$), and pegging stays the dominant rejection cause under either setup.
		\\
		\textbf{Price-space as the alternative objective.} Switching the in-engine objective back to
		price-space \eqref{eq:obj-price} changes only what each restart minimises, not how a day is
		selected or gated (\S\ref{sec:objective}). The IV-space default weights the deep-OTM wings evenly
		in vol points, whereas the price objective discounts them by their small prices. One consequence
		is specific to the IV-space default: it no longer controls relative price, so the returned price
		RMSE is large on the cheap wing cells (maximum $J_{\mathrm{px}}=120.8$, with $656$ days above
		$0.2$). Those days are tight in vol points yet far off in relative price on a near-worthless wing
		cell, which is why the gate runs in IV space.
		
		\subsection{Partial conclusions} 
			The objective is not the binding choice. Pegging is a structural
			degeneracy in the $(\kappa,\rho,\eta)$ skew subspace, and it survives a change of residual. The
			pooled SPX surface demands more skew than single-factor Heston can supply without driving one of
			$\kappa$ or $\rho$ to a corner, and the optimiser picks whichever corner. Acceptance therefore
			turns on the no-pegging clause, not on fit quality or on which residual is minimised. The levers
			that would move it are the identification fixes: anchoring $\kappa$, a soft Feller penalty, and
			liquidity weighting.
		
		\subsection{Avenues for improvement}\label{sec:improvements} 
			Acceptance is near one half of attempted days, and Feller violation is now universal on the accepted 
			set. The maturity floor has already been raised to $\tau_{\min}=14$ days and an objective wing-weighting 
			has been wired but tested null; neither moved acceptance. The remaining levers attack the $(\kappa,\rho,\eta)$ 
			degeneracy at its source. Fixing or anchoring $\kappa$ from a cross-day-robust estimate would break the 
			$\kappa\!\leftrightarrow\!\rho$ trade-off so $\rho$ stops pegging. A soft Feller penalty in the restart-selection 
			criterion, with a revisit of the $\eta$ cap, would curb the now-universal Feller violation. Warm-starting each day
			from the previous day's accepted parameters would stabilise the search. The Levenberg--Marquardt
			end criteria have not been tuned and are a further candidate.
		
			A full-sample Bates calibration is the other open item. The engine, bounds, and routing are in
			place (\S\ref{sec:jumps}, Table~\ref{tab:bates-bounds}), and the jumps give the model a direct way
			to carry the tail the single-factor vol-of-vol now overfits. Whether that relieves the
			$(\kappa,\rho,\eta)$ degeneracy and the $\eta$-driven Feller violation across the full 2012--2024
			sample is the question a full Bates run would settle.
		
			Rough volatility is the natural next paradigm, but a separate one. Empirical realised variance is
			well described by a fractional process with Hurst exponent $H\approx 0.1$ \citep{gatheral2018}, far
			from the $H=\tfrac12$ that the diffusive Heston and Bates dynamics assume. A rough model fits the
			steep short-maturity skew that single-factor Heston cannot reach without driving $\eta$ to its cap,
			so it speaks directly to the degeneracy and Feller findings above. It does not, however, fit the
			routine of this paper. The dynamics are non-Markovian, so the finite-dimensional box, the seed grid,
			and the Feller quantity have no direct analogue. There is no QuantLib engine for it. The fastest
			tractable variant, rough Heston, prices through a fractional Riccati equation
			\citep{eleuch2019} that is far heavier per evaluation than the closed-form Heston integral, and
			rough Bergomi \citep{bayer2016} prices by Monte Carlo. Multi-start calibration over thousands of
			days is then out of reach without a surrogate pricer. There is also a tension with the present
			surface: the rough skew is sharpest at the shortest maturities, exactly the tenors the pre-filter
			drops at $\tau_{\min}=14$ days \eqref{eq:prefilter}. A fair test would re-admit the short end, build
			a rough-Heston pricer or a learned surrogate, and run on a pilot window rather than the full sample.
			We leave it to separate work.

	\section{Algorithm}\label{sec:algorithm}
		\algsetup{linenosize=\small}
		Algorithm~\ref{alg:perday} states the per-day routine in the active vector $\Psi$; the driver
		applies it to every trading day and accumulates accepted rows into a single file
		(\S\ref{sec:output}). Heston is the default ($\Psi=\Phi$). The Bates fit is the same algorithm with
		$\Psi=\Theta$: the engine is the \texttt{BatesEngine} and the box $\mathcal{B}$ is the
		$\dim\Theta$-dimensional one, while the pegging test \eqref{eq:pegged} still ranges over the
		$\dim\Phi$ diffusive coordinates only.
		\begin{algorithm}[H]
			\captionsetup{font=normal}
			\caption{Per-day calibration}\label{alg:perday}
			\begin{algorithmic}[1]
				\REQUIRE day $d$ of OTM trades; rates $r(d),g(d)$; constants of Table~\ref{tab:const}
				\STATE $\mathcal{D} \leftarrow \{ i : \sigma_i>0,\ \tau_i\ge\tau_{\min}\}$ \hfill (pre-filter \eqref{eq:prefilter})
				\IF{$r(d)$ or $g(d)$ undefined on/before $d$}
				\STATE \textbf{return} $\varnothing$ \hfill (no rate)
				\ENDIF
				\STATE $S_{\mathrm{ref}} \leftarrow (\sum q_i S_i)/(\sum q_i)$; \\
				$R \leftarrow \max_i S_i / \min_i S_i - 1$; \\
				$\textsc{high\_move}\leftarrow \mathbf{1}\{R>R_{\max}\}$
				\STATE $K^\star_i \leftarrow \Delta\,\round(m_i S_{\mathrm{ref}}/\Delta)$ with $m_i=K_i/S_i$ \hfill (re-strike \eqref{eq:kstar})
				\STATE $\mathcal{T}\leftarrow$ top-$N_t$ maturities by volume;\ \ build $\mathcal{K}_\tau$ ($n_k$ per wing); \\ pivot to $\{\hat\sigma_{k,\tau}\}$, cells $\mathcal{N}$
				\IF{coverage \eqref{eq:coverage} fails}
				\STATE clear stale diagnostics;\ \textbf{return} $\varnothing$
				\ENDIF
				\STATE $\hat v \leftarrow \clip(\med(\hat\sigma)^2,[10^{-3},0.25])$;\ \ build seeds $\mathcal{S}$ \eqref{eq:seeds}
				\FOR{$s \in \mathcal{S}$}
				\STATE $\widehat\Psi_s \leftarrow$ LM minimiser of $\sum_n(\varepsilon^{\mathrm{obj}}_n)^2$ on $\mathcal{B}$ from $s$ \hfill \eqref{eq:localproblem}
				\STATE compute $J_{\mathrm{iv}}(\widehat\Psi_s)$ \hfill \eqref{eq:obj-iv}
				\ENDFOR
				\STATE $\Psi^\star \leftarrow \argmin_{s} J_{\mathrm{iv}}(\widehat\Psi_s)$ \hfill (select \eqref{eq:select})
				\IF{$J_{\mathrm{iv}}(\Psi^\star) \le \varepsilon_{\mathrm{iv}}$ \AND $\lnot\,\textsc{pegged}(\Psi^\star)$}
				\STATE write repriced diagnostics;\ \textbf{return} row$(d,S_{\mathrm{ref}},r,g,\Psi^\star,\mathcal{F},J_{\mathrm{iv}},J_{\mathrm{px}},\dots)$
				\ELSE
				\STATE clear stale diagnostics;\ \textbf{return} $\varnothing$ \hfill (rejected)
				\ENDIF
			\end{algorithmic}
		\end{algorithm}
